   324→  const [notFound, setNotFound] = useState(false);
   325→
   326→  useEffect(() => {
   327→    async function fetchCourse() {
   328→      try {
   329→        const res = await fetch('/api/courses');
   330→        if (res.ok) {
   331→          const data = await res.json();
   332→          if (data.courses && data.courses.length > 0) {
   333→            const found = data.courses.find((c: Course) => c.slug === slug);
   334→            if (found) {
   335→              setCourse(found);
   336→            } else {
   337→              const fallback = fallbackCourses.find((c) => c.slug === slug);
   338→              if (fallback) {
   339→                setCourse(fallback);
   340→              } else {
   341→                setNotFound(true);
   342→              }
   343→            }
   344→          } else {
   345→            const fallback = fallbackCourses.find((c) => c.slug === slug);
   346→            if (fallback) {
   347→              setCourse(fallback);
   348→            } else {
   349→              setNotFound(true);
   350→            }
   351→          }
   352→        } else {
   353→          const fallback = fallbackCourses.find((c) => c.slug === slug);

   355→            setCourse(fallback);
